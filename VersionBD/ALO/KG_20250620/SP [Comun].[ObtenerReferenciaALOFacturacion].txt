-- Author:		<QUEVEDO CASTILLO CARLOS JAHIR>

-- Description:	<GENERA LA REFERENCIA DE FACTURACIÓN.>
-- =============================================
ALTER PROCEDURE [Comun].[ObtenerReferenciaALOFacturacion]
	-- Add the parameters for the stored procedure here
	@pIdCatLineaNegocio			INT,
	@pIdOrden					INT,
	@pReferenciaClienteExterno	NVARCHAR(50),
	@pIdCatClienteFacturar		INT,
	@ReferenciaAloFacturar		NVARCHAR(100) OUTPUT,
	@Mensaje					NVARCHAR(100) OUTPUT
AS
BEGIN
	-- SET NOCOUNT ON added to prevent extra result sets from
	-- interfering with SELECT statements.
	SET NOCOUNT ON;

    DECLARE
		@SufijoActual	INT,
		@LenSufijo		INT,
		@UltimoSegmento NVARCHAR(10),
		@BaseReferencia NVARCHAR(100),
		@TableName		NVARCHAR(100),
		@SQLBase		NVARCHAR(MAX)

	-- DETERMINAMOS LA TABLA SEGUN LA LINEA DE NEGOCIO
	SET @TableName = 
		CASE 
			WHEN @pIdCatLineaNegocio = 1 THEN 'integracionReferencia'
			WHEN @pIdCatLineaNegocio IN (2, 3) THEN 'SLO.integracionReferencia'
			ELSE NULL
		END
	
	-- VALIDAMOS SI APLICA
	IF @TableName IS NOT NULL
	BEGIN
		-- VERIFICAMOS SI YA EXISTE LA REFERENCIA CON CLIENTE EXTERNO
		SET @SQLBase = '
		SELECT TOP 1 @ReferenciaAloFacturar = ReferenciaALO
		FROM ' + @TableName + '
		WHERE IdOrden = @pIdOrden AND ReferenciaClienteExterno = @pReferenciaClienteExterno AND ClaveClienteExterno = @pIdCatClienteFacturar
		ORDER BY IdIntReferencia DESC'
	
		EXEC sp_executesql 
			@SQLBase,
			N'@pIdOrden INT, @pReferenciaClienteExterno NVARCHAR(50), @pIdCatClienteFacturar INT, @ReferenciaAloFacturar NVARCHAR(100) OUTPUT',
			@pIdOrden,
			@pReferenciaClienteExterno,
			@pIdCatClienteFacturar,
			@ReferenciaAloFacturar OUTPUT
	
		-- DEVOLVEMOS LA REFERENCIA ALO EN CASO DE QUE YA EXISTA UNA REFERENCIA LIGADA A LA REFERENCIA CLIENTE EXTERNO
		IF @ReferenciaAloFacturar IS NOT NULL
		BEGIN
			RETURN
		END
	
		-- SI NO EXISTE UNA REFERENCIA LIGADA A LA REFERENCIA CLIENTE EXTERNO QUE SE ESTA INGRESANDO EN EL PARAMETRO, ENTONCES BUSCAMOS LA ULTIMA GENERADA PARA ESE NÚMERO DE ORDEN
		SET @SQLBase = '
		SELECT TOP 1 @ReferenciaAloFacturar = ReferenciaALO
		FROM ' + @TableName + '
		WHERE IdOrden = @pIdOrden
		ORDER BY IdIntReferencia DESC'
	
		EXEC sp_executesql 
			@SQLBase,
			N'@pIdOrden INT, @ReferenciaAloFacturar NVARCHAR(100) OUTPUT',
			@pIdOrden,
			@ReferenciaAloFacturar OUTPUT
	
		-- SI ENCONTRAMOS UNA REFERENCIA BASE, SE APLICA LÓGICA DE SUFIJO
		IF @ReferenciaAloFacturar IS NOT NULL
		BEGIN
			IF PATINDEX('%-[0-9]%', @ReferenciaAloFacturar) > 0 
			   AND ISNUMERIC(RIGHT(@ReferenciaAloFacturar, CHARINDEX('-', REVERSE(@ReferenciaAloFacturar)) - 1)) = 1
			BEGIN
				SET @LenSufijo = CHARINDEX('-', REVERSE(@ReferenciaAloFacturar)) - 1
				SET @UltimoSegmento = RIGHT(@ReferenciaAloFacturar, @LenSufijo)
	
				SET @BaseReferencia = LEFT(@ReferenciaAloFacturar, LEN(@ReferenciaAloFacturar) - @LenSufijo - 1)
				SET @SufijoActual = TRY_CAST(@UltimoSegmento AS INT)
	
				IF @SufijoActual IS NOT NULL
				BEGIN
					SET @ReferenciaAloFacturar = CONCAT(@BaseReferencia, '-', @SufijoActual + 1)
				END
				ELSE
				BEGIN
					SET @ReferenciaAloFacturar = CONCAT(@ReferenciaAloFacturar, '-1')
				END
			END
			ELSE
			BEGIN
				--SET @Mensaje = 'ERROR AL GENERAR LA REFERENCIA DE FACTURACIÓN.'
				SET @ReferenciaAloFacturar = CONCAT(@ReferenciaAloFacturar, '-1')
			END
	
			RETURN;
		END
		ELSE
		BEGIN
			RETURN;
		END
	END
	ELSE
	BEGIN
		SET @Mensaje = 'LA LÍNEA DE NEGOCIO NO TIENE CONFIGURACIÓN VÁLIDA.'
	END

END
